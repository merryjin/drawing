    <script>

        const uploadInput = document.getElementById('upload');

        const scaleInput = document.getElementById('scale');

        const scaleVal = document.getElementById('scale-val');

        const bgLayer = document.getElementById('pattern-background');


        // 1. 이미지 업로드 처리

        uploadInput.addEventListener('change', function(e) {

            const file = e.target.files[0];

            if (file) {

                const reader = new FileReader();

                reader.onload = function(event) {

                    bgLayer.style.backgroundImage = `url('${event.target.result}')`;

                };

                reader.readAsDataURL(file);

            }

        });


        // 2. 슬라이더 크기 조절 처리

        scaleInput.addEventListener('input', function(e) {

            const size = e.target.value;

            scaleVal.textContent = size;

            document.documentElement.style.setProperty('--bg-size', `${size}px`);

        });

    </script>
